### Data Explorer 界面与操作指南

Perfetto v54.0 引入了 Data Explorer，这是一个全新的无代码可视化分析界面，通过连接节点构建分析管线，使复杂的性能数据分析变得直观易用。

### 分析管线的构建与执行机制

#### 管线构建流程

```python
class PerfettoDataExplorer:
    def __init__(self):
        self.pipeline_nodes = []
        self.connections = []
    
    def add_data_source(self, source_type, query):
        """添加数据源节点"""
        node = {
            'type': 'data_source',
            'source_type': source_type,
            'query': query,
            'id': self.generate_node_id()
        }
        self.pipeline_nodes.append(node)
        return node['id']
    
    def add_transformation(self, transformation_type, params):
        """添加变换节点"""
        node = {
            'type': 'transformation',
            'transformation_type': transformation_type,
            'params': params,
            'id': self.generate_node_id()
        }
        self.pipeline_nodes.append(node)
        return node['id']
    
    def add_output(self, output_type, config):
        """添加输出节点"""
        node = {
            'type': 'output',
            'output_type': output_type,
            'config': config,
            'id': self.generate_node_id()
        }
        self.pipeline_nodes.append(node)
        return node['id']
    
    def connect_nodes(self, source_id, target_id):
        """连接节点"""
        connection = {
            'source': source_id,
            'target': target_id
        }
        self.connections.append(connection)
    
    def execute_pipeline(self, trace_data):
        """执行分析管线"""
        results = {}
        
        # 1. 执行数据源查询
        for node in self.pipeline_nodes:
            if node['type'] == 'data_source':
                data = self.execute_data_source(node, trace_data)
                results[node['id']] = data
        
        # 2. 执行变换操作
        for node in self.pipeline_nodes:
            if node['type'] == 'transformation':
                source_data = self.get_source_data(node, results)
                transformed_data = self.execute_transformation(node, source_data)
                results[node['id']] = transformed_data
        
        # 3. 执行输出操作
        output_results = []
        for node in self.pipeline_nodes:
            if node['type'] == 'output':
                source_data = self.get_source_data(node, results)
                output = self.execute_output(node, source_data)
                output_results.append(output)
        
        return output_results
    
    def build_performance_pipeline(self):
        """构建性能分析管线示例"""
        # 1. 数据源：CPU 使用情况
        cpu_source_id = self.add_data_source('sql', '''
            SELECT 
                ts,
                duration,
                thread.name
            FROM cpu
            JOIN thread ON cpu.utid = thread.utid
            WHERE thread.name LIKE 'main%'
        ''')
        
        # 2. 数据源：GPU 渲染情况
        gpu_source_id = self.add_data_source('sql', '''
            SELECT 
                ts,
                duration,
                frame_number
            FROM gpu
            WHERE draw_call_count > 0
        ''')
        
        # 3. 变换：CPU 峰值检测
        cpu_peak_id = self.add_transformation('peak_detection', {
            'field': 'duration',
            'threshold_percentile': 95
        })
        self.connect_nodes(cpu_source_id, cpu_peak_id)
        
        # 4. 变换：GPU 帧率计算
        gpu_fps_id = self.add_transformation('calculate_fps', {
            'duration_field': 'duration',
            'output_field': 'fps'
        })
        self.connect_nodes(gpu_source_id, gpu_fps_id)
        
        # 5. 输出：联合分析表格
        output_id = self.add_output('table', {
            'columns': ['ts', 'cpu_duration', 'gpu_fps', 'anomaly_score']
        })
        
        # 连接 CPU 和 GPU 数据
        self.connect_nodes(cpu_peak_id, output_id)
        self.connect_nodes(gpu_fps_id, output_id)
        
        return output_id
```

## Jank CUJ SQL 标准库模块

### relevant threads jank CUJ 模块详解

Perfetto v54.0 新增的 `relevant threads jank CUJ` 模块专门用于分析用户感知的卡顿问题，它能够自动识别与用户交互相关的线程，并计算这些线程上的卡顿指标。

#### CUJ 模块架构

```sql
-- CUJ 模块基础查询结构
SELECT 
  cuj.name,
  cuj.duration,
  cuj.start_ts,
  cuj.end_ts,
  
  -- 相关线程统计
  COUNT(DISTINCT thread.utid) as relevant_thread_count,
  SUM(cpu.duration) as total_cpu_time,
  SUM(cpu.wait_time) as total_wait_time,
  
  -- Jank 指标
  cuj.jank_score,
  cuj.fps_drops,
  cuj.frame_time_spikes
  
FROM android_cuj cuj
JOIN cpu ON cpu.ts BETWEEN cuj.start_ts AND cuj.end_ts
JOIN thread ON cpu.utid = thread.utid
WHERE cuj.name = 'scroll_cuj'
```

#### CUJ 线程识别算法

```python
class JankCUJAnalyzer:
    def __init__(self):
        self.relevant_thread_patterns = [
            'main',
            'RenderThread', 
            'AnimationThread',
            'Choreographer',
            'InputDispatcher',
            'WebView'
        ]
    
    def identify_relevant_threads(self, trace_data):
        """识别与 CUJ 相关的线程"""
        relevant_threads = []
        
        # 1. 基于线程名模式匹配
        for thread in trace_data['threads']:
            if self.is_thread_relevant(thread['name']):
                relevant_threads.append(thread)
        
        # 2. 基于线程行为分析
        behavior_based_threads = self.analyze_thread_behavior(trace_data)
        relevant_threads.extend(behavior_based_threads)
        
        # 3. 去重并返回
        return self.deduplicate_threads(relevant_threads)
    
    def is_thread_relevant(self, thread_name):
        """判断线程是否与 CUJ 相关"""
        # 检查预定义模式
        for pattern in self.relevant_thread_patterns:
            if pattern.lower() in thread_name.lower():
                return True
        
        # 检查关键事件
        if any(keyword in thread_name.lower() for keyword in 
               ['ui', 'render', 'draw', 'input', 'animation']):
            return True
        
        return False
    
    def analyze_thread_behavior(self, trace_data):
        """基于线程行为识别相关线程"""
        relevant_threads = []
        
        # 分析 CPU 使用模式
        cpu_usage_stats = self.calculate_cpu_usage_patterns(trace_data)
        
        # 分析等待模式
        wait_patterns = self.analyze_wait_patterns(trace_data)
        
        # 识别高优先级线程
        for thread in trace_data['threads']:
            thread_id = thread['utid']
            thread_name = thread['name']
            
            # 检查是否为高 CPU 使用线程
            if cpu_usage_stats.get(thread_id, 0) > 0.3:  # 30% 阈值
                relevant_threads.append(thread)
            
            # 检查是否为高等待线程
            if wait_patterns.get(thread_id, 0) > 0.2:  # 20% 阈值
                relevant_threads.append(thread)
        
        return relevant_threads
```

#### Jank 指标计算

```sql
-- Jank 指标计算示例
WITH cuj_data AS (
    SELECT 
        ts,
        duration,
        frame_number,
        CASE 
            WHEN duration > 16.67 THEN 1  -- 超过 60fps 间隔
            WHEN duration > 33.33 THEN 2  -- 超过 30fps 间隔
            WHEN duration > 66.67 THEN 3  -- 超过 15fps 间隔
            ELSE 0
        END AS jank_level
    FROM gpu
    WHERE draw_call_count > 0
),

thread_analysis AS (
    SELECT 
        thread.utid,
        thread.name,
        SUM(cpu.duration) as total_cpu_time,
        COUNT(*) as cpu_sample_count,
        SUM(cpu.wait_time) as total_wait_time
    FROM cpu
    JOIN thread ON cpu.utid = thread.utid
    GROUP BY thread.utid, thread.name
)

SELECT 
    -- 基本 CUJ 信息
    cuj.name,
    cuj.duration as cuj_duration,
    cuj.start_ts,
    
    -- Jank 统计
    SUM(CASE WHEN cuj.duration > 16.67 THEN 1 ELSE 0 END) as jank_count,
    SUM(CASE WHEN cuj.duration > 33.33 THEN 1 ELSE 0 END) as severe_jank_count,
    MAX(cuj.duration) as max_frame_time,
    
    -- 线程分析
    ta.name as primary_thread,
    ta.total_cpu_time,
    ta.total_wait_time,
    ta.total_cpu_time / NULLIF(ta.cpu_sample_count, 0) as avg_cpu_per_sample,
    
    -- 复合 Jank 分数
    (SUM(CASE WHEN cuj.duration > 16.67 THEN 1 ELSE 0 END) * 1 +
     SUM(CASE WHEN cuj.duration > 33.33 THEN 1 ELSE 0 END) * 3 +
     SUM(CASE WHEN cuj.duration > 66.67 THEN 1 ELSE 0 END) * 5) / 
     NULLIF(COUNT(*), 0) as composite_jank_score
    
FROM android_cuj cuj
JOIN cuj_data ON cuj.ts = cuj_data.ts
JOIN thread_analysis ta ON cuj.primary_thread_id = ta.utid
GROUP BY cuj.name, cuj.duration, cuj.start_ts, ta.name, 
         ta.total_cpu_time, ta.total_wait_time, ta.cpu_sample_count
```

### counter-based weighted jank metrics 实现

v54.0 引入了基于 counter 的加权 jank 指标，这种指标能够根据 jank 的严重程度和时间影响进行加权计算。

#### 加权 jank 指标算法

```sql
-- counter-based weighted jank metrics 实现
WITH frame_time_metrics AS (
    SELECT 
        ts,
        duration,
        frame_number,
        
        -- 计算相对标准差
        AVG(duration) OVER (
            ORDER BY ts 
            RANGE BETWEEN 10s PRECEDING AND CURRENT ROW
        ) as moving_avg,
        
        -- 计算标准化偏差
        CASE 
            WHEN AVG(duration) OVER (
                ORDER BY ts 
                RANGE BETWEEN 10s PRECEDING AND CURRENT ROW
            ) > 0
            THEN (duration - AVG(duration) OVER (
                ORDER BY ts 
                RANGE BETWEEN 10s PRECEDING AND CURRENT ROW
            )) / AVG(duration) OVER (
                ORDER BY ts 
                RANGE BETWEEN 10s PRECEDING AND CURRENT ROW
            )
            ELSE 0
        END as normalized_deviation,
        
        -- 计算加权分数
        CASE 
            WHEN duration > 66.67 THEN 5.0  -- 严重卡顿 (15fps)
            WHEN duration > 33.33 THEN 3.0  -- 中度卡顿 (30fps)
            WHEN duration > 16.67 THEN 1.0  -- 轻度卡顿 (60fps)
            ELSE 0.0
        END as weight_score
    FROM gpu
    WHERE draw_call_count > 0
),

time_windows AS (
    SELECT 
        FLOOR(ts / 1000000000) as window_second,
        SUM(weight_score) as total_weight,
        COUNT(*) as frame_count,
        SUM(CASE WHEN normalized_deviation > 0.5 THEN 1 ELSE 0 END) as unstable_frames
    FROM frame_time_metrics
    GROUP BY FLOOR(ts / 1000000000)
)

SELECT 
    window_second,
    total_weight,
    frame_count,
    frame_count > 0 ? total_weight / frame_count : 0 as avg_weight_per_frame,
    unstable_frames,
    frame_count > 0 ? unstable_frames * 100.0 / frame_count : 0 as unstable_percentage,
    
    -- 计算窗口级别的综合 Jank 分数
    total_weight + (unstable_percentage * 0.5) as window_jank_score,
    
    -- 计算趋势指标
    LAG(total_weight) OVER (ORDER BY window_second) as prev_window_weight,
    total_weight - LAG(total_weight) OVER (ORDER BY window_second) as weight_change
FROM time_windows
ORDER BY window_second;
```

### heap_graph_stats SQL 模块功能详解

#### 内存分配模式分析

```sql
-- 内存分配模式分析
WITH allocation_patterns AS (
  SELECT 
    hgs.timestamp,
    hgs.total_allocated,
    hgs.current_heap_size,
    
    -- 计算分配速率
    LAG(hgs.total_allocated) OVER (ORDER BY hgs.timestamp) as prev_allocated,
    hgs.total_allocated - LAG(hgs.total_allocated) OVER (ORDER BY hgs.timestamp) as allocation_rate,
    
    -- 计算内存增长率
    (hgs.current_heap_size - LAG(hgs.current_heap_size) OVER (ORDER BY hgs.timestamp)) 
    / NULLIF(LAG(hgs.current_heap_size) OVER (ORDER BY hgs.timestamp), 0) * 100 as growth_rate
  FROM heap_graph_stats hgs
),

memory_pressure_events AS (
  SELECT 
    timestamp,
    CASE 
      WHEN growth_rate > 10 THEN 'HIGH_PRESSURE'
      WHEN growth_rate > 5 THEN 'MEDIUM_PRESSURE'
      ELSE 'NORMAL'
    END as pressure_level,
    
    -- 持续时间分析
    LEAD(timestamp) OVER (ORDER BY timestamp) - timestamp as duration_ms
  FROM allocation_patterns
  WHERE growth_rate > 2
)

SELECT 
  pressure_level,
  COUNT(*) as event_count,
  AVG(duration_ms) as avg_duration_ms,
  MAX(duration_ms) as max_duration_ms,
  
  -- 压力事件的分布
  percentile(50) OVER (PARTITION BY pressure_level) as median_duration,
  percentile(90) OVER (PARTITION BY pressure_level) as p90_duration,
  
  -- 生成优化建议
  CASE 
    WHEN pressure_level = 'HIGH_PRESSURE' THEN 'Consider increasing heap size or optimizing allocation patterns'
    WHEN pressure_level = 'MEDIUM_PRESSURE' THEN 'Monitor allocation patterns and consider object pooling'
    ELSE 'Normal operation'
  END as recommendation
FROM memory_pressure_events
GROUP BY pressure_level;
```

```python
class ANRIntentAnalyzer:
    def __init__(self):
        self.intent_patterns = {
            'LAUNCH': ['android.intent.action.MAIN', 'android.intent.action.LAUNCH'],
            'NAVIGATION': ['android.intent.action.VIEW', 'android.intent.action.SEARCH'],
            'BACKGROUND': ['android.intent.action.STOP', 'android.intent.action.QUIT'],
            'SERVICE': ['android.intent.action.START', 'android.intent.action.BIND']
        }
    
    def analyze_anr_by_intent(self, anr_data):
        """按 Intent 分析 ANR"""
        analysis = {
            'intent_categories': self.categorize_by_intent(anr_data),
            'component_analysis': self.analyze_components(anr_data),
            'trend_analysis': self.analyze_trends(anr_data),
            'recommendations': self.generate_recommendations(anr_data)
        }
        
        return analysis
    
    def categorize_by_intent(self, anr_data):
        """按 Intent 类别分类"""
        categories = {}
        
        for anr in anr_data:
            if not anr.get('intent'):
                continue
                
            category = self.categorize_intent(anr['intent'])
            
            if category not in categories:
                categories[category] = {
                    'count': 0,
                    'avg_duration': 0,
                    'max_duration': 0,
                    'components': set()
                }
            
            categories[category]['count'] += 1
            categories[category]['avg_duration'] += anr['duration']
            categories[category]['max_duration'] = max(
                categories[category]['max_duration'], 
                anr['duration']
            )
            categories[category]['components'].add(anr.get('component', 'unknown'))
        
        # 计算平均值
        for category in categories:
            count = categories[category]['count']
            if count > 0:
                categories[category]['avg_duration'] /= count
        
        return categories
    
    def categorize_intent(self, intent):
        """将 Intent 分类到预定义类别"""
        for category, patterns in self.intent_patterns.items():
            if any(pattern in intent for pattern in patterns):
                return category
        
        return 'OTHER'
    
    def analyze_components(self, anr_data):
        """分析组件相关的 ANR"""
        component_stats = {}
        
        for anr in anr_data:
            component = anr.get('component', 'unknown')
            
            if component not in component_stats:
                component_stats[component] = {
                    'anr_count': 0,
                    'avg_duration': 0,
                    'intents': set(),
                    'severity_distribution': {'CRITICAL': 0, 'SEVERE': 0, 'MODERATE': 0, 'MINOR': 0}
                }
            
            component_stats[component]['anr_count'] += 1
            component_stats[component]['avg_duration'] += anr['duration']
            component_stats[component]['intents'].add(anr.get('intent', 'unknown'))
            
            # 更新严重程度分布
            severity = self.categorize_severity(anr['duration'])
            component_stats[component]['severity_distribution'][severity] += 1
        
        # 计算平均值
        for component in component_stats:
            count = component_stats[component]['anr_count']
            if count > 0:
                component_stats[component]['avg_duration'] /= count
        
        return component_stats
    
    def generate_recommendations(self, anr_data):
        """生成 ANR 优化建议"""
        recommendations = []
        
        # 按 Intent 类别生成建议
        intent_categories = self.categorize_by_intent(anr_data)
        
        for category, stats in intent_categories.items():
            if stats['count'] > 10:  # 高频 ANR 类别
                recommendations.append({
                    'category': category,
                    'priority': 'high',
                    'recommendation': f"优化 {category} 类别的 Intent 处理逻辑，平均 ANR 持续时间 {stats['avg_duration']:.0f}ms",
                    'action_items': [
                        '检查主线程耗时操作',
                        '优化数据处理逻辑',
                        '考虑异步处理'
                    ]
                })
        
        # 按组件生成建议
        component_stats = self.analyze_components(anr_data)
        
        for component, stats in component_stats.items():
            if stats['anr_count'] > 5:  # 高频 ANR 组件
                recommendations.append({
                    'category': 'component',
                    'component': component,
                    'priority': 'medium',
                    'recommendation': f"优化组件 {component} 的 ANR 问题，严重程度分布: {stats['severity_distribution']}",
                    'action_items': [
                        '检查组件生命周期',
                        '优化状态管理',
                        '减少 UI 操作复杂度'
                    ]
                })
        
        return recommendations
```

### ANR 根因分析效率提升

```sql
-- ANR 根因分析查询
WITH anr_root_cause_analysis AS (
  SELECT 
    anr.pid,
    anr.process_name,
    anr.intent,
    anr.component,
    anr.anr_time,
    anr.duration,
    anr.main_thread_stack,
    
    -- 识别 ANR 类型
    CASE 
      WHEN anr.main_thread_state = 'BLOCKED' THEN 'BLOCKED_ANR'
      WHEN anr.main_thread_state = 'WAITING' THEN 'WAITING_ANR'
      WHEN anr.main_thread_state = 'TIMED_WAITING' THEN 'TIMED_WAITING_ANR'
      ELSE 'UNKNOWN_ANR'
    END as anr_type,
    
    -- 分析堆栈中的关键方法
    REGEXP_MATCH(anr.main_thread_stack, 
                 '(android\.os\.Handler\.handleMessage|' ||
                 'android\.app\.Activity\.performResume|' ||
                 'android\.app\.ActivityThread\.H\.handleMessage|' ||
                 'com\.android\.internal\.os\.ZygoteInit\.main)') as key_methods,
    
    -- 计算线程等待时间
    thread.wait_time as main_thread_wait_time,
    
    -- 关联的 CPU 使用情况
    cpu.duration as main_thread_cpu_time,
    cpu.sample_count as main_thread_cpu_samples
  FROM android_anrs anr
  JOIN thread ON anr.pid = thread.pid AND thread.name = 'main'
  LEFT JOIN cpu ON thread.utid = cpu.utid
             AND cpu.ts BETWEEN anr.anr_time - 5000000000 AND anr.anr_time
)

SELECT 
  anr_type,
  intent,
  component,
  COUNT(*) as anr_count,
  AVG(duration) as avg_duration,
  MAX(duration) as max_duration,
  AVG(main_thread_wait_time) as avg_wait_time,
  AVG(main_thread_cpu_time) as avg_cpu_time,
  
  -- 根因分析
  CASE 
    WHEN main_thread_wait_time > duration * 0.8 THEN 'WAITING_DOMINANT'
    WHEN main_thread_cpu_time > duration * 0.6 THEN 'CPU_INTENSIVE'
    WHEN key_methods IS NOT NULL THEN 'SPECIFIC_BLOCK'
    ELSE 'GENERIC_BLOCK'
  END as root_cause_category,
  
  -- 生成优先级
  CASE 
    WHEN anr_count > 10 AND avg_duration > 3000 THEN 'CRITICAL'
    WHEN anr_count > 5 OR avg_duration > 2000 THEN 'HIGH'
    WHEN anr_count > 2 OR avg_duration > 1000 THEN 'MEDIUM'
    ELSE 'LOW'
  END as priority_level,
  
  -- 优化建议
  CASE 
    WHEN main_thread_wait_time > duration * 0.8 THEN 
      'Focus on reducing main thread waiting time. Check for lock contention or network operations.'
    WHEN main_thread_cpu_time > duration * 0.6 THEN 
      'Main thread CPU intensive. Consider offloading heavy computations to background threads.'
    WHEN key_methods IS NOT NULL THEN 
      'Identified specific blocking methods. Optimize these methods or use alternative approaches.'
    ELSE 
      'Generic ANR. Profile main thread execution to identify bottlenecks.'
  END as optimization_recommendation
  
FROM anr_root_cause_analysis
GROUP BY anr_type, intent, component
ORDER BY anr_count DESC, avg_duration DESC;
```

```python
class FlameGraphGenerator:
    def __init__(self):
        self.colors = {
            'ui': '#ff6b6b',
            'render': '#4ecdc4',
            'network': '#45b7d1',
            'database': '#96ceb4',
            'cpu': '#feca57',
            'io': '#ff9ff3'
        }
    
    def generate_flamegraph_data(self, trace_data):
        """生成火焰图数据"""
        flamegraph_entries = []
        
        # 按线程处理
        for thread_id, thread_data in trace_data['threads'].items():
            thread_name = thread_data['name']
            
            # 处理每个栈帧
            for stack_trace in thread_data['stack_traces']:
                # 解析栈帧
                frames = self.parse_stack_trace(stack_trace)
                
                # 生成火焰图条目
                flamegraph_entry = self.create_flamegraph_entry(
                    thread_name, frames, thread_data['duration']
                )
                
                flamegraph_entries.append(flamegraph_entry)
        
        return flamegraph_entries
    
    def parse_stack_trace(self, stack_trace):
        """解析栈帧"""
        # 移除方法签名中的参数信息
        clean_trace = re.sub(r'\([^)]*\)', '', stack_trace)
        
        # 按包名分割栈帧
        frames = clean_trace.split(' at ')
        
        # 过滤空帧并反转（从栈顶到栈底）
        return [frame.strip() for frame in frames if frame.strip()][::-1]
    
    def create_flamegraph_entry(self, thread_name, frames, duration):
        """创建火焰图条目"""
        # 生成唯一标识
        frame_id = ';'.join(frames)
        
        # 确定颜色类别
        color_category = self.categorize_frame(frames[0] if frames else 'unknown')
        
        return {
            'id': frame_id,
            'name': thread_name,
            'frames': frames,
            'duration': duration,
            'color': self.colors.get(color_category, '#888888'),
            'level': len(frames) - 1
        }
    
    def categorize_frame(self, frame):
        """分类栈帧到颜色类别"""
        if any(keyword in frame.lower() for keyword in ['ui', 'view', 'draw', 'layout']):
            return 'ui'
        elif any(keyword in frame.lower() for keyword in ['render', 'canvas', 'gl']):
            return 'render'
        elif any(keyword in frame.lower() for keyword in ['network', 'http', 'socket']):
            return 'network'
        elif any(keyword in frame.lower() for keyword in ['database', 'sql', 'cursor']):
            return 'database'
        elif any(keyword in frame.lower() for keyword in ['compute', 'algorithm', 'math']):
            return 'cpu'
        else:
            return 'other'
```

```sql
-- snap-to-boundaries 时间选择查询
SELECT

  selected_start,
  selected_end,
  selected_end - selected_start as selected_duration,
  
  -- 自动吸附的边界
  nearest_start_boundary,
  nearest_end_boundary,
  nearest_end_boundary - nearest_start_boundary as snapped_duration,
  
  -- 吸附精度
```

#### 精确时间测量工具

```python
class TimeBoundarySnapper:
    def __init__(self):
        self.boundary_threshold_ns = 1000000  # 1ms 阈值
    
    def snap_to_boundaries(self, selection_start, selection_end, trace_data):
        """将选择范围吸附到最近的边界"""
        snapped_result = {
            'original_start': selection_start,
            'original_end': selection_end,
            'snapped_start': selection_start,
            'snapped_end': selection_end,
            'boundary_type': 'NONE',
            'accuracy_improvement': 0
        }
        
        # 查找最近的 start 边界
        nearest_start = self.find_nearest_boundary(
            selection_start, 'start', trace_data
        )
        
        # 查找最近的 end 边界
        nearest_end = self.find_nearest_boundary(
            selection_end, 'end', trace_data
        )
        
        # 计算吸附精度改善
        original_duration = selection_end - selection_start
        snapped_duration = nearest_end - nearest_start
        
        accuracy_improvement = abs(original_duration - snapped_duration)
        
        # 应用吸附
        if nearest_start and nearest_end:
            snapped_result.update({
                'snapped_start': nearest_start,
                'snapped_end': nearest_end,
                'boundary_type': 'BOTH',
                'accuracy_improvement': accuracy_improvement
            })
        
        return snapped_result
    
    def find_nearest_boundary(self, target_time, boundary_type, trace_data):
        """查找最近的边界"""
        min_distance = float('inf')
        nearest_boundary = None
        
        for slice in trace_data['slices']:
            boundary_time = None
            
            if boundary_type == 'start':
                boundary_time = slice['ts']
            else:  # end
                boundary_time = slice['ts'] + slice['dur']
            
            if boundary_time:
                distance = abs(boundary_time - target_time)
                
                if distance < min_distance and distance <= self.boundary_threshold_ns:
                    min_distance = distance
                    nearest_boundary = boundary_time
        
        return nearest_boundary
```

```python
class PivotTableAnalyzer:
    def __init__(self):
        self.default_dimensions = {
            'time': ['hour', 'minute', 'second'],
            'thread': ['name', 'priority', 'state'],
            'process': ['name', 'importance', 'oom_adj'],
            'resource': ['category', 'type', 'usage']
        }
    
    def create_pivot_table(self, trace_data, row_dimensions, column_dimensions, 
                          value_field, aggregation_func='sum'):
        """创建透视表"""
        pivot_data = {
            'dimensions': {
                'rows': row_dimensions,
                'columns': column_dimensions,
                'values': value_field
            },
            'data': {},
            'aggregations': {},
            'summary': {}
        }
        
        # 按维度分组数据
        grouped_data = self.group_by_dimensions(
            trace_data, row_dimensions, column_dimensions
        )
        
        # 计算透视数据
        for row_key, row_data in grouped_data.items():
            for col_key, col_data in row_data.items():
                value = self.aggregate_values(col_data, value_field, aggregation_func)
                
                pivot_key = (row_key, col_key)
                pivot_data['data'][pivot_key] = value
        
        # 计算汇总信息
        pivot_data['summary'] = self.calculate_summary_statistics(pivot_data)
        
        return pivot_data
    
    def group_by_dimensions(self, trace_data, row_dimensions, column_dimensions):
        """按维度分组数据"""
        grouped = {}
        
        for entry in trace_data:
            # 生成行键
            row_key = tuple(entry[dim] for dim in row_dimensions)
            
            # 生成列键
            col_key = tuple(entry[dim] for dim in column_dimensions)
            
            # 初始化分组结构
            if row_key not in grouped:
                grouped[row_key] = {}
            
            if col_key not in grouped[row_key]:
                grouped[row_key][col_key] = []
            
            grouped[row_key][col_key].append(entry)
        
        return grouped
    
    def aggregate_values(self, data_points, value_field, func):
        """聚合数值"""
        values = [dp.get(value_field, 0) for dp in data_points]
        
        if func == 'sum':
            return sum(values)
        elif func == 'avg':
            return sum(values) / len(values) if values else 0
        elif func == 'count':
            return len(values)
        elif func == 'max':
            return max(values) if values else 0
        elif func == 'min':
            return min(values) if values else 0
        else:
            return sum(values)
    
    def calculate_summary_statistics(self, pivot_data):
        """计算汇总统计"""
        all_values = list(pivot_data['data'].values())
        
        return {
            'total_values': len(all_values),
            'sum': sum(all_values),
            'avg': sum(all_values) / len(all_values) if all_values else 0,
            'max': max(all_values) if all_values else 0,
            'min': min(all_values) if all_values else 0,
            'std_dev': self.calculate_std_dev(all_values)
        }
    
    def calculate_std_dev(self, values):
        """计算标准差"""
        if not values:
            return 0
        
        mean = sum(values) / len(values)
        variance = sum((x - mean) ** 2 for x in values) / len(values)
        return variance ** 0.5
```

```python
class GlobFilterEngine:
    def __init__(self):
        self.pattern_cache = {}
    
    def create_glob_filter(self, pattern):
        """创建 glob 过滤器"""
        if pattern in self.pattern_cache:
            return self.pattern_cache[pattern]
        
        # 将 glob 模式转换为正则表达式
        regex_pattern = self.glob_to_regex(pattern)
        compiled_pattern = re.compile(regex_pattern)
        
        self.pattern_cache[pattern] = compiled_pattern
        return compiled_pattern
    
    def glob_to_regex(self, pattern):
        """将 glob 模式转换为正则表达式"""
        # 转义正则特殊字符
        regex = re.escape(pattern)
        
        # 将 glob 通配符转换为正则
        regex = regex.replace(r'\*', '.*')  # * 匹配任意字符
        regex = regex.replace(r'\?', '.')   # ? 匹配单个字符
        
        # 支持字符类 [abc]
        regex = re.sub(r'\[([^]]+)\]', r'[\1]', regex)
        
        # 支持量词 {n}, {n,m}, {n,}
        regex = re.sub(r'\{(\d+)\}', r'{\1}', regex)
        regex = re.sub(r'\{(\d+),(\d*)\}', r'{\1,\2}', regex)
        
        return f'^{regex}$'
    
    def filter_with_patterns(self, data, patterns):
        """使用多个模式过滤数据"""
        filtered_data = []
        
        for item in data:
            matches = []
            
            for pattern in patterns:
                filter_func = self.create_glob_filter(pattern)
                matches.append(filter_func.search(item) is not None)
            
            # 如果至少一个模式匹配，则保留
            if any(matches):
                filtered_data.append(item)
        
        return filtered_data
    
    def search_by_category(self, data, category_mapping):
        """按类别搜索"""
        results = {}
        
        for category, patterns in category_mapping.items():
            category_results = self.filter_with_patterns(data, patterns)
            results[category] = category_results
        
        return results
```
